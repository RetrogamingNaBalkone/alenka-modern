cmake_minimum_required(VERSION 3.16)
project(alenka-modern VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SDL2 CONFIG REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_target(alenka-assets ALL
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_resources.py" "${CMAKE_CURRENT_BINARY_DIR}/data"
    VERBATIM)
add_library(alenka-original src/original.cpp src/trainer.cpp)
target_include_directories(alenka-original PUBLIC src)
add_executable(alenka src/main.cpp)
if(WIN32)
    set_target_properties(alenka PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_libraries(alenka PRIVATE user32)
    enable_language(RC)
    target_sources(alenka PRIVATE resources/alenka.rc)
    target_include_directories(alenka PRIVATE resources)
    if(MSVC)
        # The RC already embeds our complete process manifest as resource 1.
        target_link_options(alenka PRIVATE /MANIFEST:NO)
    endif()
endif()
target_link_libraries(alenka PRIVATE alenka-original SDL2::SDL2)
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE alenka-original)
add_executable(trainer-trace tests/trainer_trace.cpp)
target_link_libraries(trainer-trace PRIVATE alenka-original)
add_executable(trainer-tests tests/trainer_tests.cpp)
target_link_libraries(trainer-tests PRIVATE alenka-original)
enable_testing()
add_test(NAME original-resources COMMAND core-tests "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA")
add_test(NAME original-course COMMAND trainer-tests "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA")
file(GLOB ALENKA_SCENARIOS "${CMAKE_CURRENT_SOURCE_DIR}/tests/scenarios/*.trace")
foreach(SCENARIO IN LISTS ALENKA_SCENARIOS)
    get_filename_component(SCENARIO_NAME "${SCENARIO}" NAME_WE)
    add_test(NAME "behavior-${SCENARIO_NAME}" COMMAND trainer-trace "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA" "${SCENARIO}")
endforeach()
install(TARGETS alenka RUNTIME DESTINATION .)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/data/" DESTINATION data)
install(FILES resources/README.txt DESTINATION .)
if(WIN32)
    install(FILES $<TARGET_FILE:SDL2::SDL2> DESTINATION .)
endif()
