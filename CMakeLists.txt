cmake_minimum_required(VERSION 3.10)
project(alenka-modern VERSION 2.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
endif()
find_package(SDL2 2.0.8 CONFIG QUIET)
if(NOT TARGET SDL2::SDL2)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2_COMPAT REQUIRED IMPORTED_TARGET sdl2>=2.0.8)
    add_library(SDL2::SDL2 INTERFACE IMPORTED)
    set_property(TARGET SDL2::SDL2 PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::SDL2_COMPAT)
endif()
# Ubuntu 18.04 ships CMake 3.10 and Python 3.6.
if(CMAKE_VERSION VERSION_LESS 3.12)
    find_package(PythonInterp 3.6 REQUIRED)
    set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
else()
    find_package(Python3 3.6 COMPONENTS Interpreter REQUIRED)
endif()

# GCC 8 provides std::filesystem in a separate library. Probe the selected
# compiler/sysroot rather than imposing GNU linker flags on other platforms.
include(CheckCXXSourceCompiles)
set(_filesystem_probe "#include <filesystem>\nint main() { return std::filesystem::exists(std::filesystem::path(\".\")); }")
check_cxx_source_compiles("${_filesystem_probe}" ALENKA_FILESYSTEM_NATIVE)
if(NOT ALENKA_FILESYSTEM_NATIVE)
    set(CMAKE_REQUIRED_LIBRARIES stdc++fs)
    check_cxx_source_compiles("${_filesystem_probe}" ALENKA_FILESYSTEM_STDCXXFS)
    unset(CMAKE_REQUIRED_LIBRARIES)
    if(NOT ALENKA_FILESYSTEM_STDCXXFS)
        message(FATAL_ERROR "C++17 std::filesystem is required. On Ubuntu 18.04 install g++-8 and configure with CXX=g++-8.")
    endif()
endif()
add_custom_target(alenka-assets ALL
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_resources.py" "${CMAKE_CURRENT_BINARY_DIR}/data"
    VERBATIM)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/english_course.hpp"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_course.py"
        "${CMAKE_CURRENT_BINARY_DIR}/generated/english_course.hpp"
    DEPENDS tools/embed_course.py resources/lessons-en.txt resources/eng-button.txt
    VERBATIM)
add_library(alenka-original src/original.cpp src/trainer.cpp src/profile.cpp src/session.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/generated/english_course.hpp")
target_include_directories(alenka-original PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(alenka-original PUBLIC src)
if(NOT ALENKA_FILESYSTEM_NATIVE)
    target_link_libraries(alenka-original PUBLIC stdc++fs)
endif()
add_executable(alenka src/main.cpp)
if(WIN32)
    set_target_properties(alenka PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_libraries(alenka PRIVATE user32)
    enable_language(RC)
    target_sources(alenka PRIVATE resources/alenka.rc)
    target_include_directories(alenka PRIVATE resources)
    if(MSVC)
        # The RC already embeds our complete process manifest as resource 1.
        set_property(TARGET alenka APPEND_STRING PROPERTY LINK_FLAGS " /MANIFEST:NO")
    endif()
endif()
target_link_libraries(alenka PRIVATE alenka-original SDL2::SDL2)
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE alenka-original)
add_executable(trainer-trace tests/trainer_trace.cpp)
target_link_libraries(trainer-trace PRIVATE alenka-original)
add_executable(trainer-tests tests/trainer_tests.cpp)
target_link_libraries(trainer-tests PRIVATE alenka-original)
add_executable(profile-tests tests/profile_tests.cpp)
target_link_libraries(profile-tests PRIVATE alenka-original)
add_executable(audio-tests tests/audio_tests.cpp)
target_include_directories(audio-tests PRIVATE src)
target_link_libraries(audio-tests PRIVATE SDL2::SDL2)
enable_testing()
add_test(NAME english-resources COMMAND "${Python3_EXECUTABLE}" -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py")
add_test(NAME audio-playback COMMAND audio-tests)
add_test(NAME original-resources COMMAND core-tests "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA")
add_test(NAME original-course COMMAND trainer-tests "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA")
add_test(NAME profiles COMMAND profile-tests "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA")
add_test(NAME profile-ui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_ui.py"
    $<TARGET_FILE:profile-tests> "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA")
file(GLOB ALENKA_SCENARIOS "${CMAKE_CURRENT_SOURCE_DIR}/tests/scenarios/*.trace")
foreach(SCENARIO IN LISTS ALENKA_SCENARIOS)
    get_filename_component(SCENARIO_NAME "${SCENARIO}" NAME_WE)
    add_test(NAME "behavior-${SCENARIO_NAME}" COMMAND trainer-trace "${CMAKE_CURRENT_SOURCE_DIR}/original/ALENKA" "${SCENARIO}")
endforeach()
install(TARGETS alenka RUNTIME DESTINATION .)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/data/" DESTINATION data)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    install(FILES resources/README-linux.txt DESTINATION . RENAME README.txt)
    install(FILES LICENSE DESTINATION .)
else()
    install(FILES resources/README.txt DESTINATION .)
endif()
if(WIN32)
    install(FILES $<TARGET_FILE:SDL2::SDL2> DESTINATION .)
endif()
